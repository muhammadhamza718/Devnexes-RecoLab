## 1. Quality Check Result Schema

### 1.1 QualityCheckResult Data Structure

```python
from dataclasses import dataclass, field
from datetime import datetime
from typing import List, Optional
from enum import Enum

class SeverityLevel(Enum):
    CRITICAL = "CRITICAL"
    HIGH = "HIGH"
    MEDIUM = "MEDIUM"
    LOW = "LOW"

class CheckCategory(Enum):
    FUNCTIONAL = "FUNCTIONAL"
    TECHNICAL = "TECHNICAL"
    SECURITY = "SECURITY"
    PERFORMANCE = "PERFORMANCE"
    DOCUMENTATION = "DOCUMENTATION"
    REPOSITORY = "REPOSITORY"
    INTEGRATION = "INTEGRATION"
    SUBMISSION = "SUBMISSION"

@dataclass
class QualityCheckResult:
    """Result of a single quality check."""
    check_id: str
    check_name: str
    category: CheckCategory
    status: str  # PASS, FAIL, WARNING
    timestamp: datetime
    evidence: str
    findings: List[str] = field(default_factory=list)
    recommendations: List[str] = field(default_factory=list)
    reviewer: str = "Agent-1"  # Default reviewer
    severity: SeverityLevel = SeverityLevel.MEDIUM
    location: Optional[str] = None  # File/component reference
    impact: Optional[str] = None  # Description of impact
    
    def to_dict(self) -> dict:
        """Convert to dictionary for JSON serialization."""
        return {
            "check_id": self.check_id,
            "check_name": self.check_name,
            "category": self.category.value,
            "status": self.status,
            "timestamp": self.timestamp.isoformat(),
            "evidence": self.evidence,
            "findings": self.findings,
            "recommendations": self.recommendations,
            "reviewer": self.reviewer,
            "severity": self.severity.value,
            "location": self.location,
            "impact": self.impact
        }
```

### 1.2 Validation Rules

**QualityCheckResult Validation:**
- `check_id`: Must be unique, format "QA-XXX" or "SP-XXX"
- `check_name`: Must be descriptive, max 100 characters
- `category`: Must be valid CheckCategory enum value
- `status`: Must be one of "PASS", "FAIL", "WARNING"
- `timestamp`: Must be valid datetime, not in future
- `evidence`: Must be non-empty string, max 500 characters
- `findings`: List must contain at least one item if status is FAIL or WARNING
- `recommendations`: List must contain at least one actionable recommendation
- `severity`: Must be valid SeverityLevel enum value
- `location`: Must be valid file path if provided

## 2. Agent Report Schema

### 2.1 AgentReport Data Structure

```python
@dataclass
class AgentReport:
    """Report generated by a verification agent."""
    agent_id: str
    agent_name: str
    validation_timestamp: datetime
    overall_status: str  # PASS, FAIL, WARNING
    confidence_score: float  # 0.0 to 1.0
    summary: str
    findings: List[QualityCheckResult] = field(default_factory=list)
    recommendations: List[str] = field(default_factory=list)
    evidence: List[str] = field(default_factory=list)
    execution_time_seconds: float = 0.0
    resources_used: dict = field(default_factory=dict)
    
    def to_dict(self) -> dict:
        """Convert to dictionary for JSON serialization."""
        return {
            "agent_id": self.agent_id,
            "agent_name": self.agent_name,
            "validation_timestamp": self.validation_timestamp.isoformat(),
            "overall_status": self.overall_status,
            "confidence_score": self.confidence_score,
            "summary": self.summary,
            "findings": [f.to_dict() for f in self.findings],
            "recommendations": self.recommendations,
            "evidence": self.evidence,
            "execution_time_seconds": self.execution_time_seconds,
            "resources_used": self.resources_used
        }
```

### 2.2 Validation Rules

**AgentReport Validation:**
- `agent_id`: Must match pattern "Agent-N" where N is 1-5
- `agent_name`: Must be descriptive, max 50 characters
- `validation_timestamp`: Must be valid datetime, not in future
- `overall_status`: Must be one of "PASS", "FAIL", "WARNING"
- `confidence_score`: Must be between 0.0 and 1.0
- `summary`: Must be non-empty string, max 500 characters
- `findings`: List must contain at least one item
- `recommendations`: List must contain at least one actionable recommendation
- `evidence`: List must contain at least one evidence reference
- `execution_time_seconds`: Must be positive number, max 3600 (1 hour)

## 3. Quality Gate Schema

### 3.1 QualityGateResult Data Structure

```python
class GateStatus(Enum):
    PASS = "PASS"
    FAIL = "FAIL"
    PENDING = "PENDING"
    OVERRIDE = "OVERRIDE"

class RiskLevel(Enum):
    LOW = "LOW"
    MEDIUM = "MEDIUM"
    HIGH = "HIGH"
    CRITICAL = "CRITICAL"

@dataclass
class GateCriterion:
    """Single criterion within a quality gate."""
    criterion_id: str
    criterion_name: str
    description: str
    status: str  # PASS, FAIL, PENDING
    evidence: str
    weight: float = 1.0  # Weight for scoring
    
@dataclass
class QualityGateResult:
    """Result of quality gate evaluation."""
    gate_id: str
    gate_name: str
    status: GateStatus
    risk_level: RiskLevel
    criteria_results: List[GateCriterion] = field(default_factory=list)
    overall_score: float = 0.0  # 0.0 to 1.0
    override_justification: Optional[str] = None
    timestamp: datetime = field(default_factory=datetime.now)
    blocking: bool = False  # Whether this gate blocks submission
    
    def to_dict(self) -> dict:
        """Convert to dictionary for JSON serialization."""
        return {
            "gate_id": self.gate_id,
            "gate_name": self.gate_name,
            "status": self.status.value,
            "risk_level": self.risk_level.value,
            "criteria_results": [
                {
                    "criterion_id": c.criterion_id,
                    "criterion_name": c.criterion_name,
                    "description": c.description,
                    "status": c.status,
                    "evidence": c.evidence,
                    "weight": c.weight
                }
                for c in self.criteria_results
            ],
            "overall_score": self.overall_score,
            "override_justification": self.override_justification,
            "timestamp": self.timestamp.isoformat(),
            "blocking": self.blocking
        }
```

```python
class EvidenceType(Enum):

    TEST_RESULT = "TEST_RESULT"
    METRIC = "METRIC"
    DOCUMENT = "DOCUMENT"
    VIDEO = "VIDEO"

@dataclass
class Evidence:
    """Single evidence item for submission package."""
    evidence_id: str
    evidence_type: EvidenceType
    category: str  # e.g., "UI", "Testing", "Performance"
    description: str
    file_path: str
    timestamp: datetime
    validator: str  # Which agent/validator collected this
    file_size_bytes: int = 0
    checksum: Optional[str] = None  # SHA256 checksum for integrity
    
    def to_dict(self) -> dict:
        """Convert to dictionary for JSON serialization."""
        return {
            "evidence_id": self.evidence_id,
            "evidence_type": self.evidence_type.value,
            "category": self.category,
            "description": self.description,
            "file_path": self.file_path,
            "timestamp": self.timestamp.isoformat(),
            "validator": self.validator,
            "file_size_bytes": self.file_size_bytes,
            "checksum": self.checksum
        }
```

### 9.1 Quality Gate Evaluation Logic

```python
def evaluate_quality_gate(gate: QualityGateResult) -> str:
    """Evaluate quality gate and return status."""
    # Calculate weighted score
    total_weight = sum(c.weight for c in gate.criteria_results)
    weighted_score = sum(
        c.weight * (1.0 if c.status == "PASS" else 0.0)
        for c in gate.criteria_results
    )
    gate.overall_score = weighted_score / total_weight if total_weight > 0 else 0.0
    
    # Determine status based on score and criteria
    if gate.overall_score >= 0.8 and all(c.status == "PASS" for c in gate.criteria_results):
        gate.status = GateStatus.PASS
        gate.blocking = False
    elif gate.overall_score >= 0.6 and gate.override_justification:
        gate.status = GateStatus.OVERRIDE
        gate.blocking = False
    else:
        gate.status = GateStatus.FAIL
        gate.blocking = True
    
    # Determine risk level
    if gate.overall_score >= 0.9:
        gate.risk_level = RiskLevel.LOW
    elif gate.overall_score >= 0.7:
        gate.risk_level = RiskLevel.MEDIUM
    elif gate.overall_score >= 0.5:
        gate.risk_level = RiskLevel.HIGH
    else:
        gate.risk_level = RiskLevel.CRITICAL
    
    return gate.status.value
```

### 9.2 Submission Readiness Evaluation Logic

```python
def evaluate_submission_readiness(quality_gates: dict) -> str:
    """Evaluate overall submission readiness based on quality gates."""
    blocked_gates = [gate for gate in quality_gates.values() if gate.blocking]
    
    if blocked_gates:
        return "BLOCKED"
    elif all(gate.status == GateStatus.PASS for gate in quality_gates.values()):
        return "READY"
    else:
        return "NOT_READY"
```
